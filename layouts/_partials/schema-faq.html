{{- /* schema-faq.html Emits FAQPage JSON-LD derived from the page's H2 sections. Opt-in per page via front matter: schema: faq # single value schema: ["faq", ...] # list form also accepted Each top-level

is treated as a Question and the HTML that follows it (up to the next

) as the accepted Answer. HTML is reduced to plain text. */ -}} {{- $schema := .Params.schema | default "" -}} {{- $isFaq := false -}} {{- if reflect.IsSlice $schema -}} {{- if in $schema "faq" -}}{{- $isFaq = true -}}{{- end -}} {{- else if eq $schema "faq" -}} {{- $isFaq = true -}} {{- end -}} {{- if $isFaq -}} {{- $qa := slice -}} {{- $parts := split .Content "" -}} {{- $head := index $sub 0 -}} {{- $question := index (split $head ">") 1 | plainify | htmlUnescape | replaceRE `\s+` " " | strings.TrimSpace -}} {{- $answer := "" -}} {{- if gt (len $sub) 1 -}} {{- $answer = index $sub 1 | plainify | htmlUnescape | replaceRE `\s+` " " | strings.TrimSpace -}} {{- end -}} {{- if and $question $answer -}} {{- $qa = $qa | append (dict "@type" "Question" "name" $question "acceptedAnswer" (dict "@type" "Answer" "text" $answer) ) -}} {{- end -}} {{- end -}} {{- end -}} {{- if gt (len $qa) 0 -}} {{- $ld := dict "@context" "https://schema.org" "@type" "FAQPage" "url" .Permalink "inLanguage" "en-US" "mainEntity" $qa -}} {{- printf "" ($ld | jsonify (dict "indent" " ")) | safeHTML -}} {{- end -}} {{- end -}}
