{{- /* Custom page description for meta tags. Resolution order: 1. Bibliography pages: " - <abstract>" (or "<title> by <authors>"). 2. Any other page: front matter `description`, then the page `.Summary`, then a generated description for taxonomy/term pages, then the site default description. The final string is normalized (plain text, single-spaced) and capped at ~160 characters so it renders well in search results. */ -}} {{- $desc := "" -}} {{- if eq .Type "bibliography" -}} {{- /* Bibliography page: construct description from title + abstract/authors */ -}} {{- $title := .Title | plainify | htmlUnescape | replaceRE "\\s+" " " | strings.TrimSpace -}} {{- $abstract := "" -}} {{- with .Params.abstract -}} {{- $abstract = . | plainify | htmlUnescape | replaceRE "\\s+" " " | strings.TrimSpace -}} {{- end -}} {{- if $abstract -}} {{- $desc = printf "%s - %s" $title $abstract -}} {{- else -}} {{- $authors := "" -}} {{- with .Params.authors -}} {{- if reflect.IsSlice . -}} {{- $authors = delimit . ", " -}} {{- else -}} {{- $authors = . -}} {{- end -}} {{- end -}} {{- if $authors -}} {{- $desc = printf "%s by %s" $title $authors -}} {{- else -}} {{- $desc = $title -}} {{- end -}} {{- end -}} {{- else -}} {{- with .Description -}} {{- $desc = . -}} {{- else -}} {{- with .Summary -}} {{- $desc = . -}} {{- else -}} {{- /* Generated descriptions for auto-created taxonomy pages */ -}} {{- if eq .Kind "term" -}} {{- if eq .Data.Singular "author" -}} {{- $desc = printf "Medley Interlisp bibliography entries authored by %s." .Title -}} {{- else if eq .Data.Singular "item_type" -}} {{- $desc = printf "Medley Interlisp bibliography entries of type \"%s\"." .Title -}} {{- else -}} {{- $desc = printf "Medley Interlisp bibliography entries for %s." .Title -}} {{- end -}} {{- else if eq .Kind "taxonomy" -}} {{- $desc = printf "Browse an index of the %s taxonomy used to organize entries in the Medley Interlisp bibliography, with links to related publications, authors, and resources." (.Title | lower) -}} {{- else -}} {{- $desc = site.Params.description -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- /* Normalize and cap length for SERP display. */ -}} {{- $desc = $desc | plainify | htmlUnescape | replaceRE "\\s+" " " | strings.TrimSpace -}} {{- if gt (len $desc) 160 -}} {{- $desc = $desc | truncate 157 -}} {{- end -}} {{- $desc -}}
