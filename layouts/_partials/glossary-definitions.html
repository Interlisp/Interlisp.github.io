{{- /* Build a map of glossary anchor IDs to short, plain-text definitions. The glossary uses definition-list syntax: BBN : Bolt, Beranek, and Newman... Hugo renders that as:
BBN
Bolt, Beranek, and Newman...
An entry may provide a custom tooltip instead of using its first
: Interlisp-D The rendered content is parsed once per build by partialCached in the render-link hook. Only the first
for each term is used, unless the entry has a data-tooltip attribute. */ -}} {{- $defs := dict -}} {{- $maxChars := 200 -}} {{- with site.Params.glossary -}} {{- $maxChars = .tooltip_max_chars | default 200 -}} {{- end -}} {{- with site.GetPage "/history/glossary" -}} {{- $entries := findRE `(?s)
]*>.*?` .Content -}} {{- range $entries -}} {{- $entry := . -}} {{- $id := replaceRE `(?s)^
]*>.*$` "$1" $entry -}} {{- $text := "" -}} {{- if strings.Contains $entry "data-tooltip=" -}} {{- $custom := replaceRE `(?s)^
]*\sdata-tooltip="([^"]*)"[^>]*>.*$` "$1" $entry -}} {{- $text = $custom | plainify | htmlUnescape | replaceRE `\s+` " " | replaceRE `\s+([,.;:!?])` "$1" | chomp -}} {{- else -}} {{- $definition := replaceRE `(?s)^.*?
(.*)
$` "$1" $entry -}} {{- $text = $definition | plainify | htmlUnescape | replaceRE `\s+` " " | replaceRE `\s+([,.;:!?])` "$1" | chomp | truncate $maxChars -}} {{- end -}} {{- if $text -}} {{- $defs = merge $defs (dict $id $text) -}} {{- else -}} {{- warnf "Glossary entry #%s produced empty tooltip text" $id -}} {{- end -}} {{- end -}} {{- end -}} {{- return $defs -}}
